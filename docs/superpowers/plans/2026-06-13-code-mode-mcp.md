# Code Mode MCP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Cloudflare-style Code Mode search+execute tools to both MCP servers — `camunda_search`/`camunda_execute` in `aikit-mcp.ts` for the full Camunda 8 REST API, and `sdk_search`/`sdk_execute` in `mcp-server.ts` for `@bpmnkit/core` — so AI agents can discover and call any operation without enumerating 180+ tools.

**Architecture:** Phase 1 adds Camunda API Code Mode to `aikit-mcp.ts` using `isolated-vm` (credentials live in scope, stricter isolation needed). Phase 2 adds SDK Code Mode to `mcp-server.ts` using Node's built-in `vm` module (pure computation, no credentials, keeps mcp-server.ts bundleable with esbuild). Both phases share the same two-tool pattern: a `search` tool for schema introspection and an `execute` tool for live execution. All new tools coexist alongside existing ones.

**Tech Stack:** `isolated-vm` (v4.x, native addon), Node `vm` module, `@bpmnkit/api`'s `CamundaClient`, `@bpmnkit/core` SDK functions, Vitest.

```ts
// apps/proxy/src/sandbox.ts
import ivm from "isolated-vm"

export type HostFunction = (...args: unknown[]) => Promise<unknown>

export interface SandboxContext {
  /** JSON-serializable values injected as named globals */
  data?: Record<string, unknown>
  /** Async host functions exposed as ivm.References */
  functions?: Record<string, HostFunction>
  /** JS injected before user code (e.g. proxy builders) */
  bootstrap?: string
}

export async function runSandboxed(
  code: string,
  ctx: SandboxContext,
  timeoutMs = 5000,
): Promise<unknown> {
  const isolate = new ivm.Isolate({ memoryLimit: 64 })
  try {
    const context = await isolate.createContext()
    const jail = context.global
    await jail.set("global", jail.derefInto())

    for (const [key, value] of Object.entries(ctx.data ?? {})) {
      await jail.set(key, new ivm.ExternalCopy(value).copyInto())
    }

    for (const [key, fn] of Object.entries(ctx.functions ?? {})) {
      await jail.set(
        key,
        new ivm.Reference(async (...args: unknown[]) => {
          const result = await fn(...args)
          return new ivm.ExternalCopy(result).copyInto()
        }),
      )
    }

    const fullCode = ctx.bootstrap ? `${ctx.bootstrap}\n${code}` : code

    return await context.evalClosure(
      `return (async function() {\n${fullCode}\n})()`,
      [],
      { result: { promise: true, copy: true }, timeout: timeoutMs },
    )
  } finally {
    isolate.dispose()
  }
}
```

The Camunda API has ~30 resource groups and ~180 operations. This script parses the auto-generated `resources.ts` to produce a spec object with full parameter and return type info.

- [ ] **Step 1: Write the generator script**

```js
// scripts/generate-camunda-spec.mjs
import { readFileSync, writeFileSync } from "node:fs"

const source = readFileSync("packages/api/src/generated/resources.ts", "utf-8")

// ── Step 1: resource key → class name from CamundaClient body ────────────────
const clientBody = source.match(
  /export class CamundaClient extends CamundaBaseClient \{([\s\S]*?)(?=\n\n  constructor|\n\n  [a-z])/
)?.[1] ?? ""
if (!clientBody) throw new Error("Could not find CamundaClient body")

const resourceMap = {}
const propRe = /readonly (\w+):\s*(\w+Resource)/g
let m
while ((m = propRe.exec(clientBody)) !== null) resourceMap[m[1]] = m[2]

console.log(`Found ${Object.keys(resourceMap).length} resource groups`)

// ── Step 2: extract method specs from each resource class ────────────────────
const spec = {}
for (const [resourceKey, className] of Object.entries(resourceMap)) {
  const classRe = new RegExp(
    `export class ${className} extends ResourceBase \\{([\\s\\S]*?)(?=\\nexport class|$)`
  )
  const classBody = source.match(classRe)?.[1]
  if (!classBody) { console.warn(`  ⚠ class ${className} not found`); continue }

  spec[resourceKey] = {}

  // Match JSDoc block immediately followed by async method signature
  const methodRe =
    /\/\*\*([\s\S]*?)\*\/\s+async (\w+)\(([^)]*)\)[^:]*:\s*Promise<([^>]+)>/g
  let mm
  while ((mm = methodRe.exec(classBody)) !== null) {
    const [, jsdoc, methodName, rawParams, returnType] = mm
    const lines = jsdoc
      .split("\n")
      .map((l) => l.replace(/^\s*\*\s?/, "").trim())
      .filter(Boolean)
    const descLines = lines.filter((l) => !l.startsWith("@"))
    const endpoint =
      lines.find((l) => l.startsWith("@see"))?.replace("@see ", "") ?? ""
    const params = rawParams.trim()
      ? rawParams.trim().replace(/Types\./g, "")
      : "none"

    spec[resourceKey][methodName] = {
      description: descLines.join(" ").trim(),
      endpoint,
      params,
      returns: returnType.replace(/Types\./g, "").trim(),
    }
  }
}

const methodCount = Object.values(spec).reduce(
  (s, ms) => s + Object.keys(ms).length,
  0
)
console.log(`Extracted ${methodCount} methods`)

// ── Step 3: write TypeScript file ────────────────────────────────────────────
const ts = `// Auto-generated by scripts/generate-camunda-spec.mjs — do not edit manually.
// Re-run: node scripts/generate-camunda-spec.mjs

export interface MethodSpec {
  description: string
  endpoint: string
  params: string
  returns: string
}

export const CAMUNDA_SPEC: Record<string, Record<string, MethodSpec>> = ${JSON.stringify(spec, null, 2)} as const
`

writeFileSync("apps/proxy/src/camunda-spec.ts", ts)
console.log("Wrote apps/proxy/src/camunda-spec.ts")
```

```ts
// Bootstrap injected into the isolate — builds a camunda proxy over __dispatch.
// The proxy intercepts resource.method(args) calls and routes them to the host client.
const CAMUNDA_PROXY_BOOTSTRAP = `
const camunda = new Proxy({}, {
  get(_, resource) {
    return new Proxy({}, {
      get(_, method) {
        return async (args) => {
          const json = await __dispatch.apply(
            undefined,
            [resource, method, JSON.stringify(args ?? null)],
            { result: { promise: true, copy: true } }
          )
          return JSON.parse(json)
        }
      }
    })
  }
})
`

export async function handleCamundaExecute(code: string): Promise<string> {
  const profile = getActiveProfile()
  if (!profile) throw new Error("No active Camunda profile")

  const client = new CamundaClient(profile.config)

  const dispatch: HostFunction = async (
    resource: unknown,
    method: unknown,
    argsJson: unknown,
  ) => {
    const rc = (client as Record<string, Record<string, (a: unknown) => Promise<unknown>>>)[
      String(resource)
    ]
    if (!rc) throw new Error(`Unknown resource: ${resource}`)
    const fn = rc[String(method)]
    if (typeof fn !== "function") throw new Error(`Unknown method: ${resource}.${method}`)
    const result = await fn.call(rc, JSON.parse(String(argsJson)))
    return JSON.stringify(result ?? null)
  }

  const result = await runSandboxed(
    code,
    { bootstrap: CAMUNDA_PROXY_BOOTSTRAP, functions: { __dispatch: dispatch } },
    10000,
  )
  return JSON.stringify(result)
}
```

Therefore sdk tools use Node's built-in `vm` module. SDK functions (parse, exportXml, optimize, layout, analyzeVariables) are all synchronous pure computations — no async needed in the vm context.

`sdk-code-mode.ts` is a separate module so it can be imported in tests without triggering the mcp-server.ts readline stdio loop.

```ts
// apps/proxy/src/sdk-code-mode.ts
import vm from "node:vm"
import {
  Bpmn,
  type BpmnDefinitions,
  type BpmnDiagram,
  type BpmnProcess,
  compactify,
  expand,
  optimize,
  applyAutoLayout,
  analyzeVariableFlow,
} from "@bpmnkit/core"
import { SDK_SPEC } from "./sdk-spec.js"

// Mirrors the same helper in mcp-server.ts — builds BPMN DI shapes from a process.
function buildBpmnDiagram(proc: BpmnProcess): BpmnDiagram {
  const shapes: import("@bpmnkit/core").BpmnDiShape[] = proc.flowElements
    .filter((e): e is import("@bpmnkit/core").BpmnFlowElement & { id: string } => "id" in e)
    .map((el) => ({
      id: `${el.id}_di`,
      bpmnElement: el.id,
      bounds: { x: 0, y: 0, width: 100, height: 80 },
    }))
  return { id: `${proc.id}_plane`, bpmnElement: proc.id, shapes, edges: [] }
}

function buildSdkContext(xml?: string): Record<string, unknown> {
  return {
    xml: xml ?? "",
    sdk: {
      parse: (rawXml: unknown): string => {
        const defs = Bpmn.parse(String(rawXml))
        return JSON.stringify(compactify(defs))
      },
      exportXml: (compactJson: unknown): string => {
        const defs = expand(JSON.parse(String(compactJson)))
        const laidOut = applyAutoLayout(defs)
        return Bpmn.export(laidOut)
      },
      optimize: (compactJson: unknown): string => {
        const defs = expand(JSON.parse(String(compactJson)))
        const result = optimize(defs)
        return JSON.stringify({
          diagram: compactify(result.definitions),
          findings: result.findings,
        })
      },
      layout: (compactJson: unknown): string => {
        const defs = expand(JSON.parse(String(compactJson)))
        const laidOut = applyAutoLayout(defs)
        return JSON.stringify(compactify(laidOut))
      },
      analyzeVariables: (compactJson: unknown): string => {
        const defs = expand(JSON.parse(String(compactJson)))
        return JSON.stringify(analyzeVariableFlow(defs))
      },
    },
  }
}

function runInVm(code: string, ctx: Record<string, unknown>, timeoutMs: number): unknown {
  const context = vm.createContext(ctx)
  try {
    const result = vm.runInContext(`(function(){\n${code}\n})()`, context, {
      timeout: timeoutMs,
    })
    return result
  } catch (err) {
    throw new Error(
      `Code execution failed: ${err instanceof Error ? err.message : String(err)}`,
    )
  }
}

export function handleSdkSearch(code: string): Promise<string> {
  const result = runInVm(code, { spec: SDK_SPEC }, 5000)
  return Promise.resolve(JSON.stringify(result))
}

export function handleSdkExecute(code: string, xml?: string): Promise<string> {
  const result = runInVm(code, buildSdkContext(xml), 10000)
  return Promise.resolve(JSON.stringify(result))
}
```
